{{- /* The fields a template needs to render one cask, from a raw entry in data/catalog.json. Keeps the style-name mapping and the sample fallback in one place instead of in every layout. A font for another script has a line in it and then the Latin sample on a second line; `line` is the first, for rows of one line. */ -}} {{- $c := .cask -}} {{- $sample := $c.sample | default .sampleText -}} {{- $styleNames := dict "monospaced" "Fixed Width" "serif" "Serif" "sansSerif" "Sans Serif" "display" "Display" "handwriting" "Handwriting" -}} {{- return (dict "slug" (strings.TrimPrefix "font-" $c.token) "token" $c.token "name" $c.name "sample" $sample "line" (index (split $sample "\n") 0) "styleKey" $c.style "styleName" (index $styleNames $c.style | default "Sans Serif") "glyphs" $c.glyphs "installs" $c.installs "previewFile" $c.previewFile ) -}}
